(function(){
const range=document.getElementById('range');
const reticle=document.getElementById('reticle');
const blaster=document.getElementById('blaster');
const toast=document.getElementById('toast');
const message=document.getElementById('message');
const startBtn=document.getElementById('start');

const hitsEl=document.getElementById('hits');
const errorsEl=document.getElementById('errors');
const timeEl=document.getElementById('time');

const A=Array.from({length:15},function(_,i){return String(i+1)});

function rel(){return current&&current.perfStart?performance.now()-current.perfStart:0}
function pointer(e){const r=range.getBoundingClientRect();return{x:Math.max(0,Math.min(1,(e.clientX-r.left)/r.width)),y:Math.max(0,Math.min(1,(e.clientY-r.top)/r.height)),px:e.clientX-r.left,py:e.clientY-r.top,w:r.width,h:r.height}}

function createDummy(label,pos){const el=document.createElement('div');el.className='dummy';el.style.left=(pos.x*100)+'%';el.style.top=(pos.y*100)+'%';el.innerHTML='<div class="code"></div><div class="head"></div><div class="body"></div><div class="legs"></div>';el.querySelector('.code').textContent=label;range.appendChild(el);return{id:++serial,label:label,x:pos.x,y:pos.y,spawn_ms:+rel().toFixed(3),hit_ms:null,el:el}}

function spawnNext(){const s=sequence();if(current.nextSpawn>=s.length)return;const p=safePosition(current.active);current.active.push(createDummy(s[current.nextSpawn],p));current.nextSpawn+=1}
function expectedLabel(){return sequence()[current.expectedIndex]}

function finishRound(reason){if(!current||current.finished)return;current.finished=true;current.duration_ms:+Math.min(rel(),LIMIT).toFixed(3);current.finished_at=new Date().toISOString();current.end_reason=reason;cancelAnimationFrame(raf);clearDummies();message.classList.remove('hidden');if(current.part==='A'){message.innerHTML='<h2>ROUND A COMPLETE</h2><p>Sequential Aim 종료.<br>NEXT ROUND를 눌러 Switching Aim을 시작하세요.</p>';nextBtn.disabled=false;roundEl.textContent='A COMPLETE'}else{session.completed=true;session.finished_at=new Date().toISOString();message.innerHTML='<h2>MISSION COMPLETE</h2><p>Round A/B가 완료되었습니다.<br>SEND REPORT로 행동 데이터를 분석 화면에 전송하세요.</p>';sendBtn.disabled=false;roundEl.textContent='COMPLETE'}}

function clickRange(e){if(!current||current.finished)return;const p=pointer(e);showAim(p);flashGun();const t=rel();let chosen=null;let best=Infinity;for(const target of current.active){const dx=p.px-target.x*p.w;const dy=p.py-target.y*p.h;const d=Math.hypot(dx,dy);if(d<=HIT_RADIUS&&d<best){best=d;chosen=target}}const expected=expectedLabel();if(!chosen){current.miss_clicks+=1;current.clicks.push({t_ms:+t.toFixed(3),x_norm:+p.x.toFixed(6),y_norm:+p.y.toFixed(6),type:'miss',expected:expected});showToast('MISS',false);updateHud();return}if(chosen.label!==expected){current.wrong_target_clicks+=1;current.clicks.push({t_ms:+t.toFixed(3),x_norm:+p.x.toFixed(6),y_norm:+p.y.toFixed(6),type:'wrong_target',clicked:chosen.label,expected:expected});chosen.el.classList.add('wrong');setTimeout(function(){chosen.el.classList.remove('wrong')},180);showToast('WRONG TARGET',false);updateHud();return}current.correct_hits+=1;current.clicks.push({t_ms:+t.toFixed(3),x_norm:+p.x.toFixed(6),y_norm:+p.y.toFixed(6),type:'correct',clicked:chosen.label,expected:expected});hitTarget(chosen,t);current.expectedIndex+=1;showToast('TARGET ACQUIRED',true);updateHud();if(current.expectedIndex>=sequence().length){finishRound('completed');return}spawnNext()}
function moveRange(e){const p=pointer(e);showAim(p);if(!current||current.finished)return;const t=rel();if(t-lastMove<16)return;lastMove=t;current.pointer_events.push({t_ms:+t.toFixed(3),x_norm:+p.x.toFixed(6),y_norm:+p.y.toFixed(6)})}
