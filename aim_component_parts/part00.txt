<script>
function post(type,data){
  window.parent.postMessage(Object.assign({isStreamlitMessage:true,type:type},data),"*");
}
function resize(){
  post("streamlit:setFrameHeight",{height:document.documentElement.scrollHeight});
}
function sendData(v){
  post("streamlit:setComponentValue",{value:v});
}
window.addEventListener("message",function(e){
  if(e.data&&e.data.type==="streamlit:render") setTimeout(resize,0);
});
post("streamlit:componentReady",{apiVersion:1});
setTimeout(resize,0);

(function(){
  const range=document.getElementById("range");
  const reticle=document.getElementById("reticle");
  const weapon=document.getElementById("weapon");
  const toast=document.getElementById("toast");
  const message=document.getElementById("message");
  const startBtn=document.getElementById("start");
  const nextBtn=document.getElementById("next");
  const resetBtn=document.getElementById("reset");
  const sendBtn=document.getElementById("send");
  const roundEl=document.getElementById("round");
  const hitsEl=document.getElementById("hits");
  const errorsEl=document.getElementById("errors");
  const timeEl=document.getElementById("time");
  const ruleEl=document.getElementById("rule");

  const A=Array.from({length:15},function(_,i){return String(i+1)});
  const B=["1","A","2","B","3","C","4","D","5","E","6","F","7","G","8"];
  const LIMIT=45000;
  const HIT=28;

  let session=null;
  let round=null;
  let raf=null;
  let lastMove=-Infinity;
  let serial=0;

  function uid(){
    return "aim_"+new Date().toISOString().replace(/\D/g,"").slice(0,14)+"_"+Math.random().toString(36).slice(2,8);
  }
  function seq(){return round.part==="A"?A:B;}
  function rel(){return round?performance.now()-round.t0:0;}
  function point(e){
    const r=range.getBoundingClientRect();
    return {
      x:Math.max(0,Math.min(1,(e.clientX-r.left)/r.width)),
      y:Math.max(0,Math.min(1,(e.clientY-r.top)/r.height)),
      px:e.clientX-r.left,py:e.clientY-r.top,w:r.width,h:r.height
    };
  }
  function aim(p){
    reticle.classList.remove("hidden");
    reticle.style.left=p.x*100+"%";
    reticle.style.top=p.y*100+"%";
  }
  function fire(){
    weapon.classList.remove("fire");
    void weapon.offsetWidth;
    weapon.classList.add("fire");
  }
  function pop(t,ok){
    toast.textContent=t;
    toast.style.color=ok?"#83ffe9":"#ff857d";
    toast.classList.remove("show");
    void toast.offsetWidth;
    toast.classList.add("show");
  }
  function safe(ex){
    ex=ex||[];
    for(let k=0;k<100;k++){
      const p={x:.12+Math.random()*.70,y:.20+Math.random()*.36};
      if(p.y>.48&&p.x>.62) continue;
      if(ex.every(function(q){return Math.hypot(p.x-q.x,p.y-q.y)>=.24;})) return p;
    }
    return{x:.3,y:.3};
  }
  function dummy(label,p){
    const el=document.createElement("div");
    el.className="dummy";
    el.style.left=p.x*100+"%";
    el.style.top=p.y*100+"%";
    el.innerHTML='<div class="code"></div><div class="head"></div><div class="body"></div><div class="legs"></div>';
    el.querySelector(".code").textContent=label;
    range.appendChild(el);
    return{id:++serial,label:label,x:p.x,y:p.y,el:el,spawn:+rel().toFixed(3)};
  }
  function clearTargets(){
    range.querySelectorAll(".dummy").forEach(function(el){el.remove();});
  }
  function initial(){
    const s=seq();
    const p1=safe([]);
    const p2=safe([p1]);
    round.active=[dummy(s[0],p1),dummy(s[1],p2)];
    round.next=2;
  }
  function spawn(){
    const s=seq();
    if(round.next>=s.length) return;
    const p=safe(round.active);
    round.active.push(dummy(s[round.next],p));
    round.next++;
  }
  function hud(){
    hitsEl.textContent=round.correct+" / 15";
    errorsEl.textContent=round.wrong+" / "+round.miss;
  }
  function startRound(part){
    cancelAnimationFrame(raf);
    clearTargets();
    lastMove=-Infinity;
    round={
      part:part,t0:performance.now(),started_at:new Date().toISOString(),finished:false,
      idx:0,next:0,active:[],done:[],moves:[],clicks:[],correct:0,wrong:0,miss:0
    };
    session.rounds.push(round);
    message.classList.add("hidden");
    roundEl.textContent="ROUND "+part;
    roundEl.className="good";
    ruleEl.textContent=part==="A"?"ROUND A: 1 → 2 → 3 → … → 15":"ROUND B: 1 → A → 2 → B → … → 8";
    timeEl.textContent="45.00 s";
    startBtn.disabled=true;
    nextBtn.disabled=true;
    sendBtn.disabled=true;
    hud();
    initial();
    tick();
  }
  function start(){
    session={
      schema_version:"1.2",
      session_id:uid(),
      mode:"cognitive_aim_two_target",
      started_at:new Date().toISOString(),
      rounds:[],
      completed:false
    };
    startRound("A");
  }
  function tick(){
    if(!round||round.finished) return;
    const left=Math.max(0,LIMIT-rel());
    timeEl.textContent=(left/1000).toFixed(2)+" s";
    if(left<=0){finishRound("timeout");return;}
    raf=requestAnimationFrame(tick);
  }
  function finishRound(reason){
    if(!round||round.finished) return;
    round.finished=true;
    round.duration_ms=+Math.min(rel(),LIMIT).toFixed(3);
    round.end_reason=reason;
    clearTargets();
    cancelAnimationFrame(raf);
    message.classList.remove("hidden");

    if(round.part==="A"){
      message.innerHTML="<h2>ROUND A COMPLETE</h2><p>NEXT ROUND를 눌러 Switching Aim을 시작하세요.</p>";
      nextBtn.disabled=false;
    }else{
      session.completed=true;
      session.finished_at=new Date().toISOString();
      const correct=session.rounds.reduce(function(a,r){return a+r.correct;},0);
      const wrong=session.rounds.reduce(function(a,r){return a+r.wrong;},0);
      const miss=session.rounds.reduce(function(a,r){return a+r.miss;},0);
      session.summary={
        correct_hits:correct,
        wrong_target_clicks:wrong,
        miss_clicks:miss,
        accuracy:correct/Math.max(1,correct+wrong+miss),
        total_pointer_events:session.rounds.reduce(function(a,r){return a+r.moves.length;},0)
      };
      message.innerHTML="<h2>MISSION COMPLETE</h2><p>SEND REPORT로 행동 데이터를 분석 화면에 전송하세요.</p>";
      sendBtn.disabled=false;
    }
  }

  function clickRange(e){
    if(!round||round.finished) return;
    const p=point(e);
    aim(p);
    fire();
    const t=rel();
    let chosen=null;
    let best=Infinity;

    round.active.forEach(function(target){
      const d=Math.hypot(p.px-target.x*p.w,p.py-target.y*p.h);
      if(d<=HIT&&d<best){best=d;chosen=target;}
    });

    const expected=seq()[round.idx];

    if(!chosen){
      round.miss++;
      round.clicks.push({t_ms:+t.toFixed(3),x_norm:+p.x.toFixed(6),y_norm:+p.y.toFixed(6),type:"miss",expected:expected});
      pop("MISS",false);
      hud();
      return;
    }
    if(chosen.label!==expected){
      round.wrong++;
      round.clicks.push({t_ms:+t.toFixed(3),x_norm:+p.x.toFixed(6),y_norm:+p.y.toFixed(6),type:"wrong_target",clicked:chosen.label,expected:expected});
      chosen.el.classList.add("wrong");
      setTimeout(function(){if(chosen.el)chosen.el.classList.remove("wrong");},180);
      pop("WRONG TARGET",false);
      hud();
      return;
    }

    round.correct++;
    const hitTime=+t.toFixed(3);
    const response=+(hitTime-chosen.spawn).toFixed(3);
    round.clicks.push({
      t_ms:hitTime,x_norm:+p.x.toFixed(6),y_norm:+p.y.toFixed(6),
      type:"correct",clicked:chosen.label,expected:expected,reaction_ms:response,correct:true
    });
    round.done.push({id:chosen.id,label:chosen.label,x:chosen.x,y:chosen.y,spawn_ms:chosen.spawn,hit_ms:hitTime,response_ms:response});
    chosen.el.classList.add("hit","removing");
    round.active=round.active.filter(function(x){return x.id!==chosen.id;});
    setTimeout(function(){if(chosen.el&&chosen.el.parentNode)chosen.el.remove();},120);
    round.idx++;
    pop("TARGET ACQUIRED",true);
    hud();

    if(round.idx>=seq().length){finishRound("completed");return;}
    spawn();
  }

  function moveRange(e){
    const p=point(e);
    aim(p);
    if(!round||round.finished) return;
    const t=rel();
    if(t-lastMove<16) return;
    lastMove=t;
    round.moves.push({t_ms:+t.toFixed(3),x_norm:+p.x.toFixed(6),y_norm:+p.y.toFixed(6)});
  }

  function resetAll(){
    cancelAnimationFrame(raf);
    session=null;round=null;
    clearTargets();
    message.classList.remove("hidden");
    message.innerHTML="<h2>COGNITIVE AIM</h2><p>항상 두 개의 훈련 타깃이 등장합니다.<br>올바른 코드의 머리를 조준하고 클릭하세요.</p>";
    roundEl.textContent="STANDBY";
    hitsEl.textContent="0 / 15";
    errorsEl.textContent="0 / 0";
    timeEl.textContent="45.00 s";
    ruleEl.textContent="ROUND A: 1 → 2 → 3 → … → 15";
    startBtn.disabled=false;
    nextBtn.disabled=true;
    sendBtn.disabled=true;
    reticle.classList.add("hidden");
  }

  function sendReport(){
    if(!session||!session.completed) return;
    session.rounds.forEach(function(r){delete r.t0;delete r.active;});
    sendData(session);
    pop("REPORT SENT",true);
  }
