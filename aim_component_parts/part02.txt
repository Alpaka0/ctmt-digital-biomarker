function seqText(part){return part==='B'?'1 → A → 2 → B → 3 → C → … → 8':'1 → 2 → 3 → … → 15'}
function instructionText(part){return part==='B'?'숫자와 문자를 번갈아 순서대로 클릭하세요.':'숫자를 작은 수부터 순서대로 클릭하세요.'}
function stageRoundText(part,stage){return 'STAGE '+stage+'/'+TOTAL_STAGES+' · ROUND '+part}
function setGuide(roundText,instruction,targetText,sequenceText){const r=document.getElementById('guideRound'),i=document.getElementById('guideInstruction'),t=document.getElementById('guideTarget'),s=document.getElementById('guideSequence');if(r)r.textContent=roundText;if(i)i.textContent=instruction;if(t)t.textContent=targetText;if(s)s.textContent=sequenceText}

function advance(hitActor,t){round.idx+=1;if(round.idx>=seq().length){finishRound('completed');return}currentActor=previewActor;currentActor.role='current';currentActor.badge.className='badge current';round.currentActivatedAt=t;previewActor=hitActor;if(round.nextIdx<seq().length){hideActor(previewActor);const label=seq()[round.nextIdx++];setTimeout(()=>{if(round&&!round.finished)setActor(previewActor,label,'preview')},125)}else{hideActor(previewActor)}refreshDistractor();updateGuide()}
function actorAt(p){let chosen=null,best=Infinity;for(const a of actors){if(!a.visible||!a.hole)continue;const cx=a.hole.x*p.w,cy=(a.hole.y-a.hole.scale*.34)*p.h;const rx=a.hole.scale*p.w*.47*HIT_PAD,ry=a.hole.scale*p.h*.54*HIT_PAD;const d=((p.px-cx)/rx)**2+((p.py-cy)/ry)**2;if(d<=1&&d<best){best=d;chosen=a}}return chosen}
function animateHit(actor,kind){actor.el.classList.remove('bonk','wrong');void actor.el.offsetWidth;actor.el.classList.add(kind);setTimeout(()=>actor.el.classList.remove(kind),190)}

function startRound(part,stageIndex){cancelAnimationFrame(raf);lastMove=-Infinity;removeReport();range.style.cursor='none';hammer.style.display='none';const rr=range.getBoundingClientRect();const trialIndex=session.rounds.length+1;round={part,stage_index:stageIndex,trial_index:trialIndex,t0:performance.now(),started_at:new Date().toISOString(),finished:false,idx:0,nextIdx:0,pointer_events:[],clicks:[],spawns:[],correct_hits:0,wrong_target_clicks:0,miss_clicks:0,duration_ms:null,currentActivatedAt:0,canvas_width:Math.round(rr.width),canvas_height:Math.round(rr.height)};session.rounds.push(round);message.classList.add('hidden');roundEl.textContent=part+stageIndex+' · '+trialIndex+'/'+TOTAL_TRIALS;roundEl.className='good';ruleEl.textContent='ROUND '+part+': '+seqText(part);timeEl.textContent=(LIMIT/1000).toFixed(2)+' s';startBtn.disabled=true;nextBtn.disabled=true;sendBtn.disabled=true;updateHud();initMoles();tick()}

function median(values){const v=values.filter(Number.isFinite).slice().sort((a,b)=>a-b);if(!v.length)return null;const m=Math.floor(v.length/2);return v.length%2?v[m]:(v[m-1]+v[m])/2}
function mean(values){const v=values.filter(Number.isFinite);return v.length?v.reduce((a,b)=>a+b,0)/v.length:null}
function pathDistance(events){let d=0;for(let i=1;i<events.length;i++)d+=Math.hypot(events[i].x_norm-events[i-1].x_norm,events[i].y_norm-events[i-1].y_norm);return d}
function pathEfficiency(r){const events=r.pointer_events||[],hits=(r.clicks||[]).filter(c=>c.correct);if(!events.length||!hits.length)return null;let prevT=0,prevPoint=null,vals=[];for(const hit of hits){let pts=events.filter(p=>p.t_ms>=prevT&&p.t_ms<=hit.t_ms).map(p=>({x:p.x_norm,y:p.y_norm}));if(prevPoint)pts.unshift(prevPoint);if(!pts.length){prevT=hit.t_ms;prevPoint={x:hit.x_norm,y:hit.y_norm};continue}pts.push({x:hit.x_norm,y:hit.y_norm});let actual=0;for(let i=1;i<pts.length;i++)actual+=Math.hypot(pts[i].x-pts[i-1].x,pts[i].y-pts[i-1].y);const direct=Math.hypot(pts[pts.length-1].x-pts[0].x,pts[pts.length-1].y-pts[0].y);if(actual>0)vals.push(Math.max(0,Math.min(1,direct/actual)));prevT=hit.t_ms;prevPoint={x:hit.x_norm,y:hit.y_norm}}return mean(vals)}
function roundReport(r){const hits=(r.clicks||[]).filter(c=>c.correct),rts=hits.map(c=>+c.reaction_ms).filter(Number.isFinite),wrong=+r.wrong_target_clicks||0,miss=+r.miss_clicks||0,correct=+r.correct_hits||0,total=correct+wrong+miss;return{part:r.part,stage_index:r.stage_index,trial_index:r.trial_index,end_reason:r.end_reason,correct_hits:correct,wrong_target_clicks:wrong,miss_clicks:miss,error_count:wrong+miss,accuracy:correct/Math.max(1,total),duration_ms:+r.duration_ms||0,mean_rt_ms:mean(rts),median_rt_ms:median(rts),trajectory_distance_norm:pathDistance(r.pointer_events||[]),path_efficiency:pathEfficiency(r),pointer_events:(r.pointer_events||[]).length}}
function aggregateReports(rs,part){const arr=rs.filter(r=>r.part===part);const nums=['correct_hits','wrong_target_clicks','miss_clicks','error_count','accuracy','duration_ms','mean_rt_ms','median_rt_ms','trajectory_distance_norm','path_efficiency','pointer_events'];const out={part,trials:arr.length};for(const k of nums)out[k]=mean(arr.map(r=>+r[k]).filter(Number.isFinite));out.correct_hits_total=arr.reduce((s,r)=>s+r.correct_hits,0);out.wrong_target_clicks_total=arr.reduce((s,r)=>s+r.wrong_target_clicks,0);out.miss_clicks_total=arr.reduce((s,r)=>s+r.miss_clicks,0);return out}
function buildSessionReport(){const rs=(session.rounds||[]).map(roundReport),a=aggregateReports(rs,'A'),b=aggregateReports(rs,'B');const rtCost=Number.isFinite(a.median_rt_ms)&&Number.isFinite(b.median_rt_ms)?b.median_rt_ms-a.median_rt_ms:null;const durationCost=Number.isFinite(a.duration_ms)&&Number.isFinite(b.duration_ms)?b.duration_ms-a.duration_ms:null;const errorDelta=Number.isFinite(a.error_count)&&Number.isFinite(b.error_count)?b.error_count-a.error_count:null;const efficiencyDelta=Number.isFinite(a.path_efficiency)&&Number.isFinite(b.path_efficiency)?b.path_efficiency-a.path_efficiency:null;return{generated_at:new Date().toISOString(),overall:session.summary||{},trial_reports:rs,part_summary:{A:a,B:b},alternation_difference:{median_rt_ms:rtCost,duration_ms:durationCost,error_delta_per_trial:errorDelta,path_efficiency_delta:efficiencyDelta}}}
function nextInfo(){if(!round)return null;if(round.part==='A')return{part:'B',stage:round.stage_index};if(round.stage_index<TOTAL_STAGES)return{part:'A',stage:round.stage_index+1};return null}

function finishSession(){range.style.cursor='default';setGuide('20 TRIALS 완료','리포트 보기 버튼을 눌러 결과를 확인하세요.','완료','A × 10 + B × 10 반복 기록 완료');session.completed=true;session.finished_at=new Date().toISOString();const correct=session.rounds.reduce((a,r)=>a+r.correct_hits,0),wrong=session.rounds.reduce((a,r)=>a+r.wrong_target_clicks,0),miss=session.rounds.reduce((a,r)=>a+r.miss_clicks,0);session.summary={completed_trials:session.rounds.length,total_trials:TOTAL_TRIALS,correct_hits:correct,max_correct_targets:TOTAL_TRIALS*15,wrong_target_clicks:wrong,miss_clicks:miss,accuracy:correct/Math.max(1,correct+wrong+miss),total_pointer_events:session.rounds.reduce((a,r)=>a+r.pointer_events.length,0)};session.report=buildSessionReport();message.innerHTML='<h2>20 Trials 완료!</h2><p>Round A 10회와 Round B 10회가 모두 끝났습니다.<br><strong>리포트 보기</strong>를 눌러 반복 행동 분석 결과를 확인하세요.</p><div class="paperMap"><b>기록 완료:</b> 20 trials · trajectory · reaction time · errors · A/B difference</div>';sendBtn.disabled=false;nextBtn.disabled=true;roundEl.textContent='COMPLETE · 20/20'}
