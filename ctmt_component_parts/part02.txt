  const TARGET_RADIUS = 0.0275;
  const MAX_TRIAL_MS = 25000;

  let plan = [];
  let session = null;
  let currentTrialIndex = -1;
  let current = null;
  let rafTimer = null;
  let lastPointerMoveMs = -Infinity;

  function uid(){
    return 'ctmt_' + new Date().toISOString().replace(/\D/g,'').slice(0,14) + '_' + Math.random().toString(36).slice(2,8);
  }

  function buildPlan(){
    const p = [
      {part:'A',is_practice:true,trial_index:0,layout_index:0},
      {part:'B',is_practice:true,trial_index:0,layout_index:1}
    ];
    for(let i=0;i<20;i++){
      p.push({
        part:(i%2===0?'A':'B'),
        is_practice:false,
        trial_index:Math.floor(i/2)+1,
        layout_index:i+2
      });
    }
    return p;
  }

  function normPointer(e){
    const r = stage.getBoundingClientRect();
    return {
      x:Math.min(1,Math.max(0,(e.clientX-r.left)/r.width)),
      y:Math.min(1,Math.max(0,(e.clientY-r.top)/r.height))
    };
  }

  function nowRel(){
    return current && current._perfStart ? performance.now()-current._perfStart : 0;
  }

  function dist(a,b){ return Math.hypot(a.x-b.x,a.y-b.y); }

  function recordPointer(e){
    if(!current || !current.started || current.finished) return;
    const rel=nowRel();
    if(e.type==='pointermove' && (rel-lastPointerMoveMs)<16) return;
    if(e.type==='pointermove') lastPointerMoveMs=rel;

    const p=normPointer(e);
    current.events.push({
      t_ms:+rel.toFixed(3),
      x_norm:+p.x.toFixed(6),
      y_norm:+p.y.toFixed(6),
      event_type:e.type,
      button:('button' in e ? e.button : null),
      buttons:('buttons' in e ? e.buttons : null)
    });
    eventCountEl.textContent=current.events.length;
    updateCrosshair(p.x,p.y);

    if(e.type==='pointermove') detectTargets(p, rel);
  }

  function detectTargets(p, rel){
    if(!current || current.finished) return;
    const expected=current.targets[current.nextIndex];
    if(!expected) return;

    const d=dist(p,{x:expected.target_x_norm,y:expected.target_y_norm});
    if(d<=TARGET_RADIUS){
      expected.reached=true;
      expected.reached_at_ms=+rel.toFixed(3);
      const el=stage.querySelector(`[data-order="${expected.target_order}"]`);
      if(el) el.classList.add('hit');
      current.correct_count++;
      current.nextIndex++;
      correctCountEl.textContent=`${current.correct_count} / 15`;
      nextTargetEl.textContent=current.targets[current.nextIndex]?.target_label ?? '완료';

      if(current.nextIndex>=current.targets.length){
        finishTrial('completed');
      }
    }
  }
