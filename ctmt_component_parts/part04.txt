  function downloadBlob(filename,text,type){
    const blob=new Blob([text],{type});
    const a=document.createElement('a');
    a.href=URL.createObjectURL(blob);a.download=filename;document.body.appendChild(a);a.click();
    setTimeout(()=>{URL.revokeObjectURL(a.href);a.remove()},0);
  }
  function downloadJSON(){
    if(session) downloadBlob(`${session.session_id}.json`,JSON.stringify(session,null,2),'application/json');
  }
  function sendSessionToPython(){
    if(!session || !session.completed) return;
    const totalEvents=session.trials.reduce((a,t)=>a+t.events.length,0);
    const analysis=session.trials.filter(t=>!t.is_practice);
    sendDataToPython({
      ...session,
      summary:{
        trial_count:analysis.length,
        valid_trial_count:analysis.filter(t=>t.is_valid).length,
        total_events:totalEvents
      }
    });
    status.textContent='Python으로 전송 완료';
    status.className='small ok';
  }
